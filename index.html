`; const PORT = process.env.PORT || 3000; const ROOMS = ["geral", "palco", "mesa-de-som", "transmissao"]; const MAX_HISTORY = 80; const MAX_BODY = 4 * 1024 * 1024; // 4 MB (mensagens de voz) const history = Object.fromEntries(ROOMS.map((r) => [r, []])); const clients = new Map(); // id -> { res, room, name } function sse(res, event, data) { res.write(`event: ${event}\ndata: ${JSON.stringify(data)}\n\n`); } function broadcast(room, event, data) { for (const c of clients.values()) { if (c.room === room) sse(c.res, event, data); } } function usersIn(room) { return [...clients.values()].filter((c) => c.room === room).map((c) => c.name); } function broadcastPresence(room) { broadcast(room, "presence", { room, users: usersIn(room) }); } function cleanName(n) { return String(n || "").replace(/[\r\n<>]/g, "").trim().slice(0, 30); } function readBody(req) { return new Promise((resolve, reject) => { let size = 0; const chunks = []; req.on("data", (c) => { size += c.length; if (size > MAX_BODY) { reject(new Error("too large")); req.destroy(); return; } chunks.push(c); }); req.on("end", () => resolve(Buffer.concat(chunks).toString("utf8"))); req.on("error", reject); }); } function json(res, code, obj) { res.writeHead(code, { "Content-Type": "application/json; charset=utf-8" }); res.end(JSON.stringify(obj)); } const server = http.createServer(async (req, res) => { const url = new URL(req.url, `http://${req.headers.host}`); // ---- Stream de eventos (chat em tempo real) ---- if (req.method === "GET" && url.pathname === "/events") { const room = url.searchParams.get("room"); const name = cleanName(url.searchParams.get("name")); if (!ROOMS.includes(room) || !name) return json(res, 400, { error: "sala ou nome inválido" }); res.writeHead(200, { "Content-Type": "text/event-stream", "Cache-Control": "no-cache, no-transform", Connection: "keep-alive", "X-Accel-Buffering": "no", }); res.write("retry: 2000\n\n"); const id = crypto.randomUUID(); clients.set(id, { res, room, name }); sse(res, "history", { room, messages: history[room] }); broadcastPresence(room); req.on("close", () => { clients.delete(id); broadcastPresence(room); }); return; } // ---- Enviar mensagem ---- if (req.method === "POST" && url.pathname === "/send") { let body; try { body = JSON.parse(await readBody(req)); } catch { return json(res, 400, { error: "requisição inválida" }); } const room = body.room; const name = cleanName(body.name); if (!ROOMS.includes(room) || !name) return json(res, 400, { error: "sala ou nome inválido" }); const msg = { id: crypto.randomUUID(), room, name, time: Date.now(), type: "text", }; if (body.type === "audio") { if (typeof body.audio !== "string" || !body.audio.startsWith("data:audio/")) return json(res, 400, { error: "áudio inválido" }); msg.type = "audio"; msg.audio = body.audio; } else { const text = String(body.text || "").trim().slice(0, 1000); if (!text) return json(res, 400, { error: "mensagem vazia" }); msg.text = text; if (body.alert === true) msg.alert = true; // chamado urgente } history[room].push(msg); if (history[room].length > MAX_HISTORY) history[room].shift(); broadcast(room, "message", msg); return json(res, 200, { ok: true }); } // ---- Lista de salas ---- if (req.method === "GET" && url.pathname === "/rooms") { return json(res, 200, { rooms: ROOMS }); } // ---- Página (HTML embutido) ---- if (req.method === "GET" && (url.pathname === "/" || url.pathname === "/index.html")) { res.writeHead(200, { "Content-Type": "text/html; charset=utf-8" }); return res.end(HTML); } res.writeHead(404); res.end(); }); // Mantém conexões SSE vivas setInterval(() => { for (const c of clients.values()) c.res.write(": ping\n\n"); }, 20000); server.listen(PORT, () => { console.log(`📻 Rádio + Chat rodando em http://localhost:${PORT}`); });
